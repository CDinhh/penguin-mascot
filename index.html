Cách dùng

<script src="penguin.js"></script>
<penguin-mascot id="pg" variant="plush" size="200"></penguin-mascot>

<script>
  const pg = document.getElementById('pg');
  pg.play('happy');                  // trả lời đúng
  await pg.play('sad');              // chờ animation xong
  pg.setAttribute('state', 'think'); // hoặc đổi bằng thuộc tính
  pg.variant = 'classic';            // đổi sang bản cổ điển
</script>
